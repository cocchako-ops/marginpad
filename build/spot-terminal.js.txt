/* ─── THE TERMINAL SHELL (2026-10-24; v2 2026-10-26) ─────────────────────────────────────────────────────────

   terminal is built by MOVING those nodes into a stage (appendChild moves a node: id, listeners and the script's
   own references survive) - the /trading-api/ key manager rule, "lift it, do not retype it". The masthead is
   moved WHOLE because five of its elements are written on every repaint.
   v2 (owner: "dizajnerski uzdigni dosta ... bas cool mesto za demo spot, sa dosta features") adds blocks the
   shell BUILDS FROM DATA THE PAGE ALREADY HOLDS - window.__spot.state() exposes the lists, the portfolio, the
   board, the feed and the picked coin - so nothing here fetches a second copy of anything. The dock is five
   cells - Market, Coins, Floor, Assets, Chat (owner) - and the chart opens when a coin is tapped.
   Nothing here decides anything about money: it arranges what exists and reads it back. */

        if (++tries < 40) setTimeout(wait, 250);
        else if (was && was !== majors) was.click();
      })();

    function curCoin() { var s = ST(); return s.coin || null; }
    function watched(key) { return watchGet().some(function (w) { return w.key === key; }); }
    function paintWatchBtn() { var c = curCoin(), b = $('tWatch'); if (!b) return; var on = !!(c && picked() && watched(c.key)); b.classList.toggle('on', on); b.title = on ? T('term_watching', 'On your watchlist') : T('term_watch', 'Watch this coin'); }
    head.querySelector('#tWatch').addEventListener('click', function () {
      var c = curCoin(); if (!c || !picked()) return;
      var a = watchGet(), i = -1; a.forEach(function (w, j) { if (w.key === c.key) i = j; });
      if (i >= 0) a.splice(i, 1); else a.unshift({ key: c.key, kind: c.kind, mint: c.mint, pool: c.pool, net: c.net, sym: c.sym, name: c.name || '', logo: c.logo || '', price: c.price });
      watchSet(a); paintWatchBtn(); paintMovers(true);
      try { if (window.__mpTrack) window.__mpTrack('spot', (i >= 0 ? 'unwatch:' : 'watch:') + c.sym); } catch (e) {}
      try { if (window.mpToast) window.mpToast({ msg: (i >= 0 ? T('term_unwatched', '{sym} removed from your watchlist', { sym: c.sym }) : T('term_watched', '{sym} added to your watchlist', { sym: c.sym })), ms: 2200, key: 'spotwatch' }); } catch (e) {}
    });

    /* 9. THE BLOCKS. One painter each, driven by a 4 s read of window.__spot.state() and repainted only when the
          thing it reads changed - the page already polls the portfolio every 30 s, the feed every 30 s and the memes
          every 2 min; this adds no request. */
    var sig = {};
    function changed(k, v) { var s = JSON.stringify(v); if (sig[k] === s) return false; sig[k] = s; return true; }

    // MOVERS + WATCHLIST + CLAIM on the market view
    function chipMeme(m, tag, tagCls) { return '<button type="button" class="tchip" data-mv="m:' + esc(m.mint) + '">' + logo(m.logo, m.sym) + esc(m.sym) + (tag ? ' <small class="' + tagCls + '">' + tag + '</small>' : '') + '</button>'; }
    function paintMovers(force) {
      var s = ST(), memes = s.memeList || [], port = s.port;
      var w = watchGet();
      var claim = !!(port && port.claimedToday === false);
      var key = { n: memes.length, u: memes.length ? (memes[0].ts || 0) + ':' + memes[memes.length - 1].ts : 0, w: w.map(function (x) { return x.key; }), c: claim, h: ((s.feed && s.feed.hot) || []).slice(0, 4).map(function (x) { return x.key + x.buyers; }) };
      if (!force && !changed('movers', key)) return;
      /* a riser is a coin with a market behind it: liquidity, volume, and a move a chart can show - a brand-new pool
         printing +199,821% on its first hour is a launch curve, not a move, and the cap keeps it off the strip */
      var live = memes.filter(function (m) { return +m.liqUsd >= 20000 && +m.vol24 >= 20000 && isFinite(+m.chg24) && +m.price > 0 && +m.chg24 <= 1000 && +m.chg24 >= -99; });
      var up = live.filter(function (m) { return +m.chg24 > 0; }).sort(function (a, b) { return (+b.chg24) - (+a.chg24); }).slice(0, 4);
      var dn = live.filter(function (m) { return +m.chg24 < 0; }).sort(function (a, b) { return (+a.chg24) - (+b.chg24); }).slice(0, 3);
      // HOT = what members are buying today (the feed's own count), the FOMO half of a meme terminal
      var hot = ((s.feed && s.feed.hot) || []).filter(function (x) { return x.buyers >= 2; }).slice(0, 4);
      var h = '';
      if (claim || up.length) {
        h += '<div class="trow">' + (claim ? '<button type="button" class="tchip tclaim" data-claim="1"><svg viewBox="0 0 24 24"><path d="M12 5v14M5 12h14"/></svg>' + esc(T('term_claim_now', 'Claim $100 today')) + '</button>' : '')
          + (up.length ? '<span class="tk">' + esc(T('term_gainers', 'Risers 24h')) + '</span>' + up.map(function (m) { return chipMeme(m, pct(m.chg24), 'up'); }).join('') : '') + '</div>';
      }
      if (dn.length || hot.length) {
        h += '<div class="trow">' + (hot.length ? '<span class="tk">' + esc(T('f_hot', 'Hot today')) + '</span>' + hot.map(function (x, i) { return '<button type="button" class="tchip" data-hot="' + i + '">' + logo(x.logo, x.sym) + esc(x.sym) + ' <small class="nw">' + x.buyers + '</small></button>'; }).join('') : '')
          + (dn.length ? '<span class="tk">' + esc(T('term_losers', 'Fallers')) + '</span>' + dn.map(function (m) { return chipMeme(m, pct(m.chg24), 'dn'); }).join('') : '') + '</div>';
      }
      mstrip._hot = hot;
      if (w.length) {
        h += '<div class="trow"><span class="tk">' + esc(T('term_watchlist', 'Watching')) + '</span>' + w.map(function (x, i) { return '<button type="button" class="tchip x" data-wi="' + i + '">' + logo(x.logo, x.sym) + esc(x.sym) + '<span class="rm" data-wrm="' + i + '" aria-label="Remove">&times;</span></button>'; }).join('') + '</div>';
      }
      mstrip.innerHTML = h;
    }
    mstrip.addEventListener('click', function (e) {
      var t = e.target;
      if (t.closest('[data-claim]')) { var cb = $('claimBtn'); if (cb && !cb.disabled) cb.click(); return; }
      var rm = t.closest('[data-wrm]'); if (rm) { e.stopPropagation(); var a = watchGet(); a.splice(+rm.getAttribute('data-wrm'), 1); watchSet(a); paintMovers(true); paintWatchBtn(); return; }
      var wi = t.closest('[data-wi]'); if (wi) { var c = watchGet()[+wi.getAttribute('data-wi')]; if (c) openCoin(c); return; }
      var ht = t.closest('[data-hot]'); if (ht && mstrip._hot) { var hx = mstrip._hot[+ht.getAttribute('data-hot')]; if (hx) { openCoin(coinOfRow(hx)); try { if (window.__mpTrack) window.__mpTrack('spot', 'hot:' + hx.sym); } catch (x) {} } return; }
      var mv = t.closest('[data-mv]'); if (mv) { var k = mv.getAttribute('data-mv'), memes = ST().memeList || [], m = memes.filter(function (x) { return 'm:' + x.mint === k; })[0]; if (m) { openCoin(coinOfMeme(m)); try { if (window.__mpTrack) window.__mpTrack('spot', 'mover:' + m.sym); } catch (x) {} } }
    });

    // STATS under the chart for the picked coin
    function paintStats() {
      var s = ST(), c = s.coin; var on = picked();
      var key = c ? c.key + ':' + on : '';
      // the majors list lands AFTER the default coin is picked, so its length is part of the signature or a major's mcap/volume never appears
      if (!changed('stats', key + ':' + ((s.memeList || []).length) + ':' + ((s.majorList || []).length) + ':' + ((s.feed && s.feed.ts) || 0))) return;
      if (!c || !on) { stats.innerHTML = ''; return; }
      // the strip belongs right under the timeframes - measured once at the end of the sheet after a build, so it is placed by its anchor, not by order of insertion
      var tfs = $('tfs'); if (tfs && tfs.parentNode === sheet && stats.previousElementSibling !== tfs) tfs.insertAdjacentElement('afterend', stats);
      var it = [];
      function cell(k, v, c2) { it.push('<span>' + esc(k) + '<b' + (c2 ? ' class="' + c2 + '"' : '') + '>' + v + '</b></span>'); }
      if (c.kind === 'meme') {
        var m = (s.memeList || []).filter(function (x) { return x.mint === c.mint; })[0] || c;
        if (m.liqUsd != null) cell(T('st_liq', 'Liquidity'), fmtK(m.liqUsd));
        if (m.fdv != null) cell(T('st_fdv', 'FDV'), fmtK(m.fdv));
        if (m.vol24 != null) cell(T('st_vol', 'Volume 24h'), fmtK(m.vol24));
        if (m.chg24 != null && isFinite(+m.chg24)) cell(T('st_24h', '24h'), pct(m.chg24), cls(m.chg24));
        var hot = ((s.feed && s.feed.hot) || []).filter(function (x) { return x.key === c.key; })[0];
        if (hot && hot.buyers) cell(T('st_buyers', 'Members bought today'), String(hot.buyers));
        if (m.liqUsd != null && m.fdv) { var lf = m.liqUsd / m.fdv * 100; cell(T('st_liqfdv', 'Liq / FDV'), lf.toFixed(lf < 1 ? 2 : 1) + '%'); }
      } else {
        var mj = (s.majorList || []).filter(function (x) { return x.sym === c.sym; })[0];
        if (mj && mj.mcap) cell(T('st_mcap', 'Market cap'), fmtK(mj.mcap));
        if (mj && mj.vol24) cell(T('st_vol', 'Volume 24h'), fmtK(mj.vol24));
        if (c.chg24 != null && isFinite(+c.chg24)) cell(T('st_24h', '24h'), pct(c.chg24), cls(c.chg24));
        var hot2 = ((s.feed && s.feed.hot) || []).filter(function (x) { return x.key === c.key; })[0];
        if (hot2 && hot2.buyers) cell(T('st_buyers', 'Members bought today'), String(hot2.buyers));
      }
      stats.innerHTML = it.join('');
    }

    // COINS summary
    function paintCoins() {
      var p = ST().port; var hs = (p && p.holds) || [];
      var key = hs.map(function (h) { return h.key + ':' + h.valueUsd + ':' + h.pnlUsd; });
      if (!changed('coins', key)) return;
      if (!hs.length) { sum.innerHTML = ''; return; }
      var val = 0, pn = 0, cost = 0, best = null, worst = null;
      hs.forEach(function (h) { val += +h.valueUsd || 0; pn += +h.pnlUsd || 0; cost += +h.costUsd || 0; if (h.pnlPct != null) { if (!best || h.pnlPct > best.pnlPct) best = h; if (!worst || h.pnlPct < worst.pnlPct) worst = h; } });
      var sy = function (h) { return (h.meta && h.meta.sym) || String(h.key).replace(/^cex:/, ''); };
      sum.innerHTML = '<div><div class="k">' + esc(T('c_sum', 'Bags')) + '</div><div class="v">' + fmtUsd(val) + ' <small>' + hs.length + '</small></div></div>'
        + '<div><div class="k">' + esc(T('c_unreal', 'Unrealized P&L')) + '</div><div class="v ' + cls(pn) + '">' + sgn(pn) + ' <small>' + (cost > 0 ? pct(pn / cost * 100) : '') + '</small></div></div>'
        + (best ? '<div><div class="k">' + esc(T('c_best', 'Best')) + '</div><div class="v"><span class="up">' + pct(best.pnlPct) + '</span> <small>' + esc(sy(best)) + '</small></div></div>' : '')
        + (worst && worst !== best ? '<div><div class="k">' + esc(T('c_worst', 'Worst')) + '</div><div class="v"><span class="dn">' + pct(worst.pnlPct) + '</span> <small>' + esc(sy(worst)) + '</small></div></div>' : '');
    }

    // FLOOR: the biggest buys in the feed it already holds
    function paintFloorTop() {
      var f = ST().feed; var rows = (f && f.rows) || [];
      if (!changed('ftop', (f && f.ts) || rows.length)) return;
      var big = rows.filter(function (r) { return r.side === 'buy' && +r.usdUsd > 0; }).sort(function (a, b) { return (+b.usdUsd) - (+a.usdUsd); }).slice(0, 5);
      fstrip.innerHTML = big.length ? '<div class="trow"><span class="tk">' + esc(T('f_big', 'Biggest buys')) + '</span>' + big.map(function (r, i) { return '<button type="button" class="tchip" data-fb="' + i + '">' + logo(r.logo, r.sym) + esc(r.sym) + ' <small>' + fmtK(r.usdUsd) + '</small></button>'; }).join('') + '</div>' : '';
      fstrip._big = big;
    }
    fstrip.addEventListener('click', function (e) { var b = e.target.closest('[data-fb]'); if (!b || !fstrip._big) return; var r = fstrip._big[+b.getAttribute('data-fb')]; if (r) openCoin(coinOfRow(r)); });

    // ASSETS: curve, allocation, season
    var COLS = ['#c2f64a', '#a78bfa', '#7db4ff', '#ffd75a', '#ff8a80', '#7ff5d6', '#f0c34a'];
    function paintAcct() {
      var s = ST(), p = s.port, b = s.board;
      if (!p) return;
      var key = { e: p.equityUsd, d: p.depositedUsd, n: (p.snaps || []).length, h: (p.holds || []).map(function (h) { return h.key + ':' + h.valueUsd; }), c: p.usdtUsd, sp: p.seasonPnlUsd, r: p.season && p.season.rank, bn: b && b.ts, bt: b && b.rows && b.rows.slice(0, 3).map(function (r) { return r.who + r.pnlUsd; }) };
      if (!changed('acct', key)) return;
      // the curve: net P&L (value minus what was put in) at each daily mark, plus the live point
      var pts = (p.snaps || []).map(function (x) { return { t: x.day, v: (+x.valueUsd || 0) - (+x.depUsd || 0) }; });
      pts.push({ t: 'now', v: (+p.equityUsd || 0) - (+p.depositedUsd || 0) });
      var cv = $('tCurve'), cr = $('tCurveR');
      if (pts.length < 2) { cv.innerHTML = '<div class="tcurve-none">' + esc(T('a_curve_none', 'Your first daily mark lands after midnight UTC - the curve starts there.')) + '</div>'; cr.textContent = ''; }
      else {
        var W = 320, Hh = 84, pad = 4, min = Infinity, max = -Infinity; pts.forEach(function (q) { if (q.v < min) min = q.v; if (q.v > max) max = q.v; }); if (max - min < 1) { max += 1; min -= 1; }
        var X = function (i) { return pad + i * (W - 2 * pad) / (pts.length - 1); }, Y = function (v) { return pad + (max - v) * (Hh - 2 * pad) / (max - min); };
        var d = pts.map(function (q, i) { return (i ? 'L' : 'M') + X(i).toFixed(1) + ' ' + Y(q.v).toFixed(1); }).join(' ');
        var last = pts[pts.length - 1].v, first = pts[0].v, col = last >= first ? '#2ebd85' : '#ff6258', zy = Y(0);
        var area = d + ' L' + X(pts.length - 1).toFixed(1) + ' ' + Hh + ' L' + X(0).toFixed(1) + ' ' + Hh + ' Z';
        var gid = 'tcg' + Math.floor(Math.random() * 1e9);
        cv.innerHTML = '<svg class="tcurve" viewBox="0 0 ' + W + ' ' + Hh + '" preserveAspectRatio="none" aria-hidden="true"><defs><linearGradient id="' + gid + '" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="' + col + '" stop-opacity=".28"/><stop offset="1" stop-color="' + col + '" stop-opacity="0"/></linearGradient></defs>'
          + (zy >= 0 && zy <= Hh ? '<line x1="0" x2="' + W + '" y1="' + zy.toFixed(1) + '" y2="' + zy.toFixed(1) + '" stroke="#2c3440" stroke-dasharray="3 4"/>' : '')
          + '<path d="' + area + '" fill="url(#' + gid + ')"/><path d="' + d + '" fill="none" stroke="' + col + '" stroke-width="2" stroke-linejoin="round" stroke-linecap="round" vector-effect="non-scaling-stroke"/>'
          + '<circle cx="' + X(pts.length - 1).toFixed(1) + '" cy="' + Y(last).toFixed(1) + '" r="3" fill="' + col + '"/></svg>'
          + '<div class="tcurve-k"><span>' + esc(T('a_curve_over', 'over {n} days', { n: pts.length - 1 })) + '</span><b class="' + cls(last - first) + '">' + sgn(last - first) + '</b></div>';
        cr.textContent = T('a_curve_now', 'net {v}', { v: sgn(last) });
      }
      // allocation: cash and each bag as a share of the whole
      var al = $('tAlloc'), eq = +p.equityUsd || 0, hs = (p.holds || []).slice().sort(function (a, b2) { return (+b2.valueUsd || 0) - (+a.valueUsd || 0); });
      if (eq > 0) {
        var parts = [{ n: T('a_cash', 'Cash'), v: +p.usdtUsd || 0, c: '#5a636e' }];
        hs.slice(0, 6).forEach(function (h, i) { parts.push({ n: (h.meta && h.meta.sym) || String(h.key).replace(/^cex:/, ''), v: +h.valueUsd || 0, c: COLS[i % COLS.length] }); });
        var rest = hs.slice(6).reduce(function (t, h) { return t + (+h.valueUsd || 0); }, 0); if (rest > 0) parts.push({ n: T('a_other', 'Other'), v: rest, c: '#2c3440' });
        al.innerHTML = '<div class="talloc">' + parts.map(function (q) { return '<i style="width:' + (q.v / eq * 100).toFixed(2) + '%;background:' + q.c + '" title="' + esc(q.n) + '"></i>'; }).join('') + '</div>'
          + '<div class="talloc-l">' + parts.map(function (q) { return '<span><i style="background:' + q.c + '"></i>' + esc(q.n) + ' <b>' + Math.round(q.v / eq * 100) + '%</b></span>'; }).join('') + '</div>';
      } else al.innerHTML = '';
      // the season as a race
      var se = $('tSeason'), sr = $('tSeasonR'), sn = p.season || {}, rows = (b && b.rows) || [], pz = (b && b.prizes) || [15, 10, 5];
      var endsIn = sn.we ? fmtLeft(sn.we - Date.now()) : '';
      sr.textContent = endsIn ? T('a_ends_in', 'ends in {t}', { t: endsIn }) : '';
      var me = sn.rank ? '#' + sn.rank + ' <small>/ ' + (sn.n || 0) + '</small>' : T('a_unranked', 'unranked');
      var paid = pz.filter(function (x) { return x > 0; }).length;
      var h2 = '<div class="trace"><div class="t"><div class="k">' + esc(T('a_rank', 'Your rank')) + '</div><div class="v">' + me + '</div></div>'
        + '<div class="t"><div class="k">' + esc(T('m_season', 'Season P&L')) + '</div><div class="v ' + cls(p.seasonPnlUsd) + '">' + sgn(p.seasonPnlUsd) + '</div></div>'
        + '<div class="t"><div class="k">' + esc(T('a_pays', 'Pays')) + '</div><div class="v">' + pz.slice(0, paid).map(function (x) { return '$' + x; }).join(' ') + '</div></div></div>';
      if (rows.length) {
        h2 += rows.slice(0, 3).map(function (r) { return '<div class="tpod' + (sn.rank === r.rank ? ' me' : '') + '"><span class="rk' + (r.rank <= paid ? ' g' : '') + '">' + r.rank + '</span><span class="who"><a data-lbu="' + esc(r.who) + '">' + esc(r.who) + '</a></span><span class="pn ' + cls(r.pnlUsd) + '">' + sgn(r.pnlUsd) + '</span><span class="pz">' + (pz[r.rank - 1] > 0 ? '$' + pz[r.rank - 1] : '') + '</span></div>'; }).join('');
        var cut = rows[paid - 1];
        if (cut && (!sn.rank || sn.rank > paid)) { var need = (+cut.pnlUsd || 0) - (+p.seasonPnlUsd || 0); h2 += '<div class="tgap">' + T('a_to_prize', 'A paid place starts at <b>{v}</b> of season P&L - you are <b>{gap}</b> away.', { v: sgn(cut.pnlUsd), gap: sgn(Math.max(0, need)) }) + '</div>'; }
        else if (sn.rank && sn.rank <= paid) h2 += '<div class="tgap"><span class="up">' + esc(T('a_in_money', 'You are in a paid place - hold it to the end of the season.')) + '</span></div>';
      }
      se.innerHTML = h2;
      // the daily claim is the loop: the Assets cell carries a dot while it is still unclaimed today
      var ab = dock.querySelector('[data-tv="assets"]'); if (ab) ab.classList.toggle('dot', p.claimedToday === false && VIEW !== 'assets');
    }

    function tick() { try { paintMovers(); paintStats(); paintCoins(); paintFloorTop(); paintAcct(); paintTheses(); paintWatchBtn(); } catch (e) {} }
    tick(); setInterval(tick, 4000);
    // a pick repaints the stats at once rather than on the next tick
    try { new MutationObserver(function () { setTimeout(paintStats, 30); }).observe($('shName') || sheet, { childList: true, characterData: true, subtree: true }); } catch (e) {}

    window.__mpSpotTerm = { show: show, view: function () { return VIEW; }, picked: picked, watch: watchGet };
  }
})();
