  // a year on one screen: one square per day, one hue, five steps by rank. Tap a square, get the day.
  function calendar(host, id) {
    var m = BY[id]; var ds = daily(id); if (!ds.length) { host.innerHTML = '<div class="empty">nothing recorded</div>'; return; }
    var vals = ds.map(function (x) { return x.v; }).filter(function (x) { return x != null && x > 0; }).sort(function (a, b) { return a - b; }), q = [0.2, 0.4, 0.6, 0.8].map(function (f) { return vals[Math.floor(vals.length * f)] || 0; });
    var step = function (v) { if (v == null) return -1; if (v <= 0) return 0; for (var i = 0; i < 4; i++) if (v <= q[i]) return i + 1; return 5; };
    var first = isoWeek(ds[0].d), map = {}; ds.forEach(function (x) { map[x.d] = x.v; }); var weeks = Math.ceil((d2t(DATA.today) - d2t(first)) / 604800000) + 1, C = 13, G = 3, Lft = 26, Tp = 16, W = Lft + weeks * (C + G), H = Tp + 7 * (C + G), o = '', lastM = -1;
    ['Mon', 'Wed', 'Fri'].forEach(function (l, i) { o += '<text x="0" y="' + (Tp + (i * 2) * (C + G) + 10) + '" class="vz-x">' + l + '</text>'; });
    for (var w = 0; w < weeks; w++) { var t0 = d2t(first) + w * 604800000, mo = new Date(t0).getUTCMonth(); if (mo !== lastM) { o += '<text x="' + (Lft + w * (C + G)) + '" y="10" class="vz-x">' + MON[mo] + '</text>'; lastM = mo; } for (var d = 0; d < 7; d++) { var k = t2d(t0 + d * 86400000); if (k > DATA.today) continue; var v = map[k], s = step(v), fill = s < 0 ? 'rgba(255,255,255,.03)' : s === 0 ? 'rgba(255,255,255,.07)' : 'rgba(194,246,74,' + [0, .16, .32, .5, .72, 1][s] + ')'; o += '<rect data-day="' + k + '" x="' + (Lft + w * (C + G)) + '" y="' + (Tp + d * (C + G)) + '" width="' + C + '" height="' + C + '" rx="2.5" fill="' + fill + '"><title>' + dayLabel(k) + ': ' + fmt(m.u, v) + '</title></rect>'; } }
    host.innerHTML = '<div class="vz-cal"><svg width="' + W + '" height="' + H + '" viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="' + esc(m.l) + ' per day">' + o + '</svg></div>'; var box = host.firstChild; box.scrollLeft = box.scrollWidth;
    box.addEventListener('click', function (e) { var r = e.target.closest('[data-day]'); if (r) day(r.getAttribute('data-day')); });
  }
