/* MarginPad ops v2 client (2026-09-03). Real file (named .js.txt so wrangler ships it as TEXT, not code; node --check still parses it), served at /api/stats/asset/ops.js behind the admin cookie.
   Hash router (#section/view), sidebar built from NAV, every native view renders from JSON endpoints only; views that are
   not ported yet embed the legacy dashboard (/api/stats?legacy=1&embed=1#<tab>) so nothing is lost while porting.
   No frameworks. No emojis. */
(function () {
  'use strict';
  var $ = function (id) { return document.getElementById(id); };
  var esc = function (s) { return String(s == null ? '' : s).replace(/[<>&"]/g, function (m) { return { '<': '&lt;', '>': '&gt;', '&': '&amp;', '"': '&quot;' }[m]; }); };
  var N = function (x) { return (+x || 0).toLocaleString('en-US'); };
  var USD = function (v, d) { return '$' + (+v || 0).toLocaleString('en-US', { minimumFractionDigits: d == null ? 2 : d, maximumFractionDigits: d == null ? 2 : d }); };
  var ago = function (t) { if (!t) return '—'; var s = Math.round((Date.now() - t) / 1000); if (s < 0) s = 0; return s < 60 ? s + 's' : s < 3600 ? Math.floor(s / 60) + 'm' : s < 86400 ? Math.floor(s / 3600) + 'h' : Math.floor(s / 86400) + 'd'; };
  var dt = function (t) { try { return new Date(t).toISOString().slice(0, 16).replace('T', ' '); } catch (e) { return '—'; } };
  // Country flag as an IMAGE: emoji flags (regional-indicator pairs) render as bare letters on Windows, so the legacy
  // dashboard never showed a single flag there. flagcdn serves 20x15 PNGs; the code stays next to it as text.
  var flag = function (c) { c = String(c || '').toUpperCase(); return /^[A-Z]{2}$/.test(c) ? '<span class="cc"><img class="fl" src="https://flagcdn.com/20x15/' + c.toLowerCase() + '.png" width="20" height="15" alt="" loading="lazy" onerror="this.style.display=\'none\'">' + c + '</span>' : ''; };
  function J(u, o) { return fetch(u, o).then(function (r) { return r.json().then(function (j) { if (!r.ok && j && !j.error) j.error = 'http_' + r.status; return j; }); }); }
  function post(u, body) { return J(u, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body || {}) }); }
  var toastT = null; function toast(msg) { var t = $('toast'); t.textContent = msg; t.hidden = false; clearTimeout(toastT); toastT = setTimeout(function () { t.hidden = true; }, 3500); }
  function stamp() { $('stamp').textContent = 'updated ' + new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }); }

  // ---------- information architecture ----------
  // 8 sections instead of 24 flat tabs. `legacy` = which old tab the view embeds until it is ported natively.
  var NAV = [
    { id: 'today', title: 'Today', views: [{ id: 'overview', title: 'Overview' }] },
    { id: 'inbox', title: 'Inbox', views: [{ id: 'support', title: 'Support', legacy: 'support' }, { id: 'chat', title: 'Trader chat', legacy: 'chat' }, { id: 'mail', title: 'Email', legacy: 'mail' }, { id: 'tgbot', title: 'Telegram bot', legacy: 'tgbot' }] },
    { id: 'money', title: 'Money', views: [{ id: 'withdrawals', title: 'Withdrawals' }, { id: 'faucet', title: 'Faucet & rewards', legacy: 'rewards' }, { id: 'premium', title: 'Premium desk', legacy: 'premium' }, { id: 'revenue', title: 'Revenue', legacy: 'revenue' }, { id: 'affiliate', title: 'Affiliate', legacy: 'affiliate' }, { id: 'shop', title: 'Shop', legacy: 'shop' }] },
    { id: 'people', title: 'People', views: [{ id: 'users', title: 'Users' }, { id: 'security', title: 'Multi-account', legacy: 'security' }, { id: 'retention', title: 'Retention', legacy: 'retention' }, { id: 'journeys', title: 'Journey map', legacy: 'jmap' }, { id: 'activity', title: 'Activity', legacy: 'activity' }] },
    { id: 'trading', title: 'Trading', views: [{ id: 'live', title: 'Live trades', legacy: 'ops' }, { id: 'botapi', title: 'Bot API', legacy: 'api' }, { id: 'spot', title: 'Demo Spot', legacy: 'spot' }] },
    { id: 'growth', title: 'Growth', views: [{ id: 'seo', title: 'Traffic & SEO', legacy: 'seo' }, { id: 'funnel', title: 'Funnel', legacy: 'funnel' }, { id: 'community', title: 'Community', legacy: 'community' }] },
    { id: 'health', title: 'Health', views: [{ id: 'overview', title: 'Alerts & health' }, { id: 'perf', title: 'Performance', legacy: 'perf' }] },
    { id: 'settings', title: 'Settings', views: [{ id: 'switches', title: 'Switches' }, { id: 'rewards', title: 'Reward config', legacy: 'settings' }] }
  ];
  var VIEWS = {}; // 'section/view' -> { render(el), refresh() }
  var counts = {}; // sidebar counters from the attention list
  var cur = null, timers = [];

  function route() { var h = (location.hash || '').replace(/^#\/?/, ''); var p = h.split('/'); var sec = NAV.filter(function (s) { return s.id === p[0]; })[0] || NAV[0]; var v = sec.views.filter(function (x) { return x.id === p[1]; })[0] || sec.views[0]; return { sec: sec, view: v, key: sec.id + '/' + v.id, arg: p.slice(2).join('/') }; }
  function renderNav() { var r = route(); $('nav').innerHTML = NAV.map(function (s) { return '<div class="sec">' + esc(s.title) + '</div>' + s.views.map(function (v) { var k = s.id + '/' + v.id; return '<a href="#' + k + '" class="' + (k === r.key ? 'on' : '') + '">' + esc(v.title) + (counts[k] ? '<span class="cnt">' + counts[k] + '</span>' : '') + '</a>'; }).join(''); }).join(''); }
  function clearTimers() { timers.forEach(clearInterval); timers = []; }
  function every(ms, fn) { timers.push(setInterval(function () { if (!document.hidden) fn(); }, ms)); }
  function show() {
    var r = route(); cur = r; clearTimers(); renderNav(); $('side').classList.remove('open');
    $('crumb').innerHTML = esc(r.sec.title) + ' <small>' + esc(r.view.title) + '</small>';
    document.title = r.view.title + ' · ' + r.sec.title + ' · MarginPad ops';
    var el = $('view'); el.innerHTML = '<div class="empty">loading…</div>'; window.scrollTo(0, 0);
    el.classList.toggle('legacyview', !!r.view.legacy); paintAttn();
    var v = VIEWS[r.key];
    if (r.view.legacy) renderLegacy(el, r.view.legacy, r.view.title); else if (v) v.render(el, r.arg); else el.innerHTML = '<div class="empty">No such view.</div>';
    try { if (navigator.sendBeacon) navigator.sendBeacon('/api/admin/usage', JSON.stringify({ tab: 'v2:' + r.key })); } catch (e) {}
  }
  window.addEventListener('hashchange', show);
  $('menuBtn').addEventListener('click', function () { $('side').classList.add('open'); });
  $('sideClose').addEventListener('click', function () { $('side').classList.remove('open'); });
  $('refreshBtn').addEventListener('click', function () { show(); loadTop(); });

  // ---------- top bar: online + attention strip ----------
  var attnItems = [];
  // The strip lives on Today/Overview only (owner 2026-09-03: on the phone it ate half of every screen); everywhere else the
  // sidebar counters carry the same signal.
  function paintAttn() { var box = $('attnbar'); if (route().key !== 'today/overview') { box.innerHTML = ''; return; } box.innerHTML = attnItems.map(function (a) { var go = GO[a.go] || 'today/overview'; return '<a class="al ' + esc(a.cls) + '" href="#' + go + '">' + a.html + '</a>'; }).join(''); }
  function loadTop() {
    J('/api/stats?format=json').then(function (d) { if (d && typeof d.online === 'number') $('online').innerHTML = '<i></i><b>' + N(d.online) + '</b> online'; }).catch(function () {});
    J('/api/admin/attention').then(function (d) {
      attnItems = (d && d.items) || []; paintAttn();
      counts = {}; attnItems.forEach(function (a) { if (a.cls === 'red' || a.cls === 'amber') { var go = GO[a.go]; if (go) counts[go] = (counts[go] || 0) + 1; } }); renderNav(); stamp();
    }).catch(function () {});
  }
  var GO = { wd: 'money/withdrawals', rewards: 'money/faucet', support: 'inbox/support', stats: 'health/overview', spot: 'trading/spot', premium: 'money/premium', users: 'people/users' };

  // ---------- components ----------
  function tile(v, l, cls, sub) { return '<div class="tile ' + (cls || '') + '"><div class="v">' + v + (sub ? '<small>' + sub + '</small>' : '') + '</div><div class="l">' + l + '</div></div>'; }
  function card(title, body, sub, right) { return '<div class="card"><h2>' + title + (sub ? '<span>' + sub + '</span>' : '') + (right ? '<span class="r">' + right + '</span>' : '') + '</h2>' + body + '</div>'; }
  function dl(v, a) { a = +a || 0; v = +v || 0; if (a < 5) return ''; var p = Math.round((v - a) / a * 100); return ' <span class="' + (p >= 0 ? 'up' : 'dn') + '">' + (p >= 0 ? '&#9650;' : '&#9660;') + Math.abs(p) + '%</span>'; }
  function spark(arr, key) { if (!arr || !arr.length) return ''; var mx = 1; arr.forEach(function (x) { if (+x[key] > mx) mx = +x[key]; }); return '<div class="spark">' + arr.map(function (x, i) { return '<i class="' + (i === arr.length - 1 ? 'last' : '') + '" style="height:' + Math.max(2, (+x[key] || 0) / mx * 100) + '%" title="' + esc(x.day || x.d || '') + ' · ' + N(x[key]) + '"></i>'; }).join('') + '</div>'; }

  // ---------- legacy embed ----------
  var LEG = {}; // tab -> iframe (kept alive)
  function renderLegacy(el, tab, title) {
    el.innerHTML = '<div class="legacy" id="legacyHost"></div>'; // one dashboard from the owner's point of view: no "legacy" wording, the frame IS the view until it is ported
    var host = $('legacyHost'); var f = LEG[tab];
    if (!f) { f = document.createElement('iframe'); f.src = '/api/stats?legacy=1&embed=1#' + tab; f.setAttribute('title', title); LEG[tab] = f; }
    host.appendChild(f);
  }

          + tile(N(s.affToday), 'money clicks today', s.affToday ? '' : 'dim', y.aff != null ? 'yday ' + N(y.aff) : '') + tile(USD(s.revToday, 0), 'est. revenue today', 'dim', 'clicks x ~$0.45')

'' : 'dim', ua.activeToday != null ? N(ua.activeToday) + ' active' : '') + tile(USD(m.dispensedTodayUsd), 'faucet dispensed today', (m.dailyCapUsd && m.dispensedTodayUsd >= m.dailyCapUsd * 0.9) ? '<div class="line"><b>Yesterday</b> (' + esc(at.yday.day) + '): <b>' + N(y.uv) + '</b> visitors' + dl(y.uv, av.uv) + ' · <b>' + N(y.ret) + '</b> returning' + dl(y.ret, av.ret) + ' · <b>' + N(y.nu) + '</b> new visitors' + dl(y.nu, av.nu) + ' · <b>' + N(y.aff) + '</b> money clicks' + dl(y.aff, av.aff) + ' · <b>' + N(y.pv) + '</b> pageviews' + dl(y.pv, av.pv) + ' <span style="color:var(--faint)">· vs the 7 full days before</span></div>' : '';

        var days = (fn && (fn.days || fn.rows)) || []; var sp = days.length ? card('Last 14 days', spark(days.slice(-14), 'uv') + '<div class="line" style="margin-top:6px">visitors per day · ' + spark(days.slice(-14), 'clicks').replace('class="spark"', 'class="spark" style="height:26px"') + ' money clicks</div>') : '';
        var hl = '<div class="grid g4" style="margin-top:12px">' + tile((h.coll && h.coll.state) || '?', 'liquidation collector', (h.coll && /LIVE|OK/.test(h.coll.state)) ? '' : 'red') + tile(h.ae && h.ae.ok ? N(h.ae.evHour) + '<small>ev/h</small>' : 'down', 'analytics engine', h.ae && h.ae.ok ? 'dim' : 'red') + tile(h.users && h.users.ok ? N(h.users.active24) : '?', 'users active 24h', 'dim') + tile(h.kv ? N(h.kv.errToday) : '?', 'broken pages today', h.kv && h.kv.errToday > 3 ? 'amber' : 'dim') + '</div>';
